def oscr(score,pred,t):
 order=np.argsort(-score,kind='stable');ends=np.r_[np.flatnonzero(np.diff(score[order])!=0),len(t)-1];k=t[order]>=0;c=k&(pred[order]==t[order]);x=np.r_[0,np.cumsum(~k)[ends]/(~k).sum()];y=np.r_[0,np.cumsum(c)[ends]/k.sum()];return float(np.sum(np.diff(x)*(y[1:]+y[:-1])/2))
def metrics(pred,t,score=None,cut=None,groups=None):

  accept=score>=cut;v.update(known_coverage=float(np.mean(accept[k])),unknown_acceptance=float(np.mean(accept[~k])),known_selective_risk=float(np.mean(pred[k&accept]!=t[k&accept])) if (k&accept).any() else None,oscr=oscr(score,pred,t),threshold=float(cut))
  for g,mask in (groups or {}).items():v[g+'_unknown_acceptance']=float(np.mean(accept[mask]))

def predict(xs,y,q,nc):
 """Only labeled support and unlabeled query features enter this predictor."""
 s=np.asarray(xs,dtype=np.float64);s/=np.linalg.norm(s,axis=1,keepdims=True);centers=np.array([s[y==c].mean(0) for c in range(nc)]);centers/=np.maximum(np.linalg.norm(centers,axis=1,keepdims=True),1e-30);out={m:[] for m in P['neighbor_controls']}
 for start in range(0,len(q),512):
  z=np.asarray(q[start:start+512],dtype=np.float64);z/=np.linalg.norm(z,axis=1,keepdims=True);sim=z@s.T;nn=np.argsort(-sim,axis=1,kind='stable')[:,:3];lab=y[nn];one=np.eye(nc)[lab[:,0]].astype(np.float32);votes=(np.eye(nc)[lab].sum(1)/3).astype(np.float32);ncm=np.eye(nc)[(z@centers.T).argmax(1)].astype(np.float32);out['1nn'].append(one);out['3nn'].append(votes);out['ncm'].append(ncm)
 return {m:np.concatenate(v) for m,v in out.items()}
